"""Check changed files for exact configured credential values without printing them."""

import argparse
import json
from pathlib import Path
import subprocess

from dotenv import dotenv_values

ROOT = Path("/home/yuqian/workspace/oce")
OUT = Path(__file__).resolve().parent
parser = argparse.ArgumentParser()
parser.add_argument("label")
args = parser.parse_args()
values = set()
for source in (Path("/home/yuqian/.oce/benchmark.env"), ROOT / ".env", ROOT / ".env.local"):
    if source.is_file():
        for key, value in dotenv_values(source).items():
            if value and (key.endswith("_KEY") or key.endswith("_TOKEN") or key.endswith("_PASSWORD")):
                if value != "sk-opencontextengine":
                    values.add(value.encode())
names = set(subprocess.check_output(["git", "diff", "HEAD", "--name-only", "--diff-filter=ACMR"], cwd=ROOT, text=True).splitlines())
names.update(subprocess.check_output(["git", "ls-files", "--others", "--exclude-standard"], cwd=ROOT, text=True).splitlines())
files = [ROOT / name for name in sorted(names) if (ROOT / name).is_file()]
hits = [str(path.relative_to(ROOT)) for path in files if any(value in path.read_bytes() for value in values)]
result = {"files_checked": len(files), "configured_credential_values_checked": len(values), "matching_files": hits,
          "note": "Exact configured-value check; the documented public sk-opencontextengine default is excluded. Credential values are never emitted."}
with (OUT / f"{args.label}-credential-scan.json").open("x") as stream:
    json.dump(result, stream, indent=2)
    stream.write("\n")
print("Credential scan:", len(files), "files;", len(hits), "matching files")
assert not hits, "Configured credential bytes found in changed files"
